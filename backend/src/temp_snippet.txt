function formatWolfVoteBreakdown(room: Room, votes: Record<string, string | null>, label: string) {
  const activeWolves = getActiveWolves(room);
  const map: Record<string, string[]> = {};
  for (const wid of activeWolves) {
    const t = votes[wid];
    if (!t) continue;
    map[t] = map[t] || [];
    map[t].push(getPlayerName(room, wid));
  }
  const targets = Object.keys(map);
  if (!targets.length) return `${label}: (khÃ´ng ai chá»n)`;
  // Keep stable ordering by player name for readability.
  targets.sort((a, b) => getPlayerName(room, a).localeCompare(getPlayerName(room, b)));
  const parts = targets.map(tid => `${getPlayerName(room, tid)}  ${map[tid]!.join(", ")}`);
  return `${label}: ${parts.join("; ")}`;
}
